import { useState, useRef } from "react"
import LearnForm from "./LearnForm"

const Learn = () => {
    const [count, setCount] = useState(1)
    const [todolist, setTodo] = useState([])
    const [input, setInput] = useState('')
    const [saveInput, setSaveInput] = useState('')
    const [inputForm, setInputForm] = useState({
        namaState : '',
        umurState : 0
    }
)

    const handleClick = () => {
        console.log('count')
        setCount(count + 1)        
    }

    const handleAdd = () => {
        setTodo([{
            title : "Learning",
            desc  : "On Progress"
        },
        ...todolist
    
    ])
    }

    const renderText = () => {
        if (saveInput === '') {
            if (input === '') {
                return "Tidak ada task"
            } else {
                return "Typing"
            }
        } else {
            return saveInput
        }
    }

    const form = useRef(null)

    return (
        <div>
            <h1>Hello World</h1>
            <button onClick={handleClick}>Likes {count}</button>
            <br />
            <h1>ToDo List</h1>
            {todolist.length === 0 ? (<p>Tidak ada task</p>) : (
            <ul>
                {todolist.map((item, index) => (
                    <li key={index}>{`${item.title} - ${item.desc}`}</li>
                ))}
            </ul>

            )}
            <br />
            <p></p>
            <button onClick={handleAdd}>Add Task</button>
            <br />
            <h1>Input</h1>
                <input type="text" placeholder="Masukkan nama"
                value={input}
                onChange={(e) => {
                    setInput(e.target.value)
                }}
                />
                <button onClick={() => {
                    setInput('')
                    setSaveInput('')
                    }}
                    disabled={input === ''}
                    >Clear</button>
                <p>{saveInput === '' ? (input === '' ? "Tidak ada input" : "Typing...") : saveInput}</p>
                <button onClick={() => {
                    setSaveInput(input)
                    setInput('')
                }}
                disabled={input === ''}
                >Submit</button>
                <br />
                <p> render {renderText()}</p>
                <br />
                <h1>Form</h1>

                <form action=""
                onSubmit={ (e) =>  {
                    e.preventDefault()
                    console.log(inputForm)
                    setInputForm( {
                        namaState : e.target.nama.value,
                        umurState : e.target.umur.value
                    },
                )
                    e.target.reset()
                }}
                ref={form}

                >
                    <input name="nama" type="text" placeholder="Masukkan nama" />
                    <br />
                    <input name="umur" type="number" placeholder="Masukkan umur" />
                    <br />
                    <button onClick={() => form.current.reset}>Cancel</button>
                    <button type="submit">Submit</button>
                <p>{`Nama : ${inputForm.namaState}`}</p>
                <p>{`Umur : ${inputForm.umurState}`}</p>
                </form>




        </div>
    )
}
export default Learn
